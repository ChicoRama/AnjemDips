import supabase from '../config/supabase.js';
import { checkDriverEligibility } from '../utils/driverEligibility.js';

const ORDER_SELECT_QUERY = `
  *,
  customer:users!customer_id(id, nama, nim, no_wa),
  driver:users!driver_id(id, nama, nim, no_wa, jenis_kendaraan, plat_nomor, kapasitas_penumpang, is_online)
`;

const MAX_ACTIVE_ORDERS_PER_CUSTOMER = 3;
const PENDING_EXPIRE_MINUTES = 30;
const JADWAL_EXPIRE_MINUTES = 30;
const MAX_PENUMPANG = 8;
const MIN_ONGKIR = 2000;
const MAX_ONGKIR = 1000000;
const ACTIVE_STATUSES = ['pending', 'accepted', 'on_progress'];

const hasValue = (v) => v !== undefined && v !== null && String(v).trim() !== '';

// Driver tidak boleh melihat kontak pemesan sebelum order diterima
const hideCustomerContact = (order, user) => {
  if (user.role === 'driver' && order.status === 'pending' && order.customer) {
    return {
      ...order,
      customer: { ...order.customer, nim: null, no_wa: null },
    };
  }
  return order;
};

// POST /api/orders (Customer membuat pesanan baru)
export const createOrder = async (req, res) => {
  try {
    if (req.user.role !== 'customer') {
      return res.status(403).json({
        success: false,
        message: 'Hanya akun dengan role Customer yang dapat membuat pesanan.',
      });
    }

    const {
      jenis_layanan,
      titik_jemput,
      titik_antar,
      catatan,
      ongkir,
      jenis_kendaraan,
      jumlah_penumpang,
      waktu_berangkat,
      jadwal_at,
      nama_barang,
      perkiraan_harga_barang,
    } = req.body;

    if (
      !jenis_layanan ||
      typeof titik_jemput !== 'string' ||
      typeof titik_antar !== 'string' ||
      titik_jemput.trim() === '' ||
      titik_antar.trim() === ''
    ) {
      return res.status(400).json({
        success: false,
        message: 'Jenis layanan, titik jemput, dan titik antar wajib diisi.',
      });
    }

    if (!['anjem', 'jastip'].includes(jenis_layanan)) {
      return res.status(400).json({
        success: false,
        message: 'Jenis layanan harus berupa "anjem" atau "jastip".',
      });
    }

    if (titik_jemput.trim().length > 200 || titik_antar.trim().length > 200) {
      return res.status(400).json({
        success: false,
        message: 'Titik jemput/antar maksimal 200 karakter.',
      });
    }

    if (typeof catatan === 'string' && catatan.trim().length > 500) {
      return res.status(400).json({
        success: false,
        message: 'Catatan maksimal 500 karakter.',
      });
    }

    // Jenis kendaraan (opsional)
    let cleanJenisKendaraan = null;
    if (hasValue(jenis_kendaraan)) {
      const normalizedVehicle = String(jenis_kendaraan).trim().toLowerCase();
      if (!['motor', 'mobil'].includes(normalizedVehicle)) {
        return res.status(400).json({
          success: false,
          message: 'Jenis kendaraan harus berupa "motor" atau "mobil".',
        });
      }
      cleanJenisKendaraan = normalizedVehicle;
    }

    // Jumlah penumpang: hanya relevan untuk AnJem (JasTip tidak mengirimnya)
    let cleanJumlahPenumpang = null;
    if (jenis_layanan === 'anjem') {
      if (hasValue(jumlah_penumpang)) {
        const parsedPenumpang = Number(jumlah_penumpang);
        if (!Number.isInteger(parsedPenumpang) || parsedPenumpang < 1) {
          return res.status(400).json({
            success: false,
            message: 'Jumlah penumpang harus berupa angka minimal 1.',
          });
        }
        cleanJumlahPenumpang = parsedPenumpang;
      } else {
        cleanJumlahPenumpang = 1;
      }

      if (cleanJenisKendaraan === 'motor' && cleanJumlahPenumpang > 1) {
        return res.status(400).json({
          success: false,
          message: 'Motor hanya dapat mengangkut 1 penumpang.',
        });
      }

      if (cleanJumlahPenumpang > MAX_PENUMPANG) {
        return res.status(400).json({
          success: false,
          message: `Jumlah penumpang maksimal ${MAX_PENUMPANG} orang.`,
        });
      }
    }

    // Waktu berangkat
    let cleanWaktuBerangkat = null;
    if (hasValue(waktu_berangkat)) {
      const normalizedWaktu = String(waktu_berangkat).trim().toLowerCase();
      if (!['sekarang', 'jadwal'].includes(normalizedWaktu)) {
        return res.status(400).json({
          success: false,
          message: 'Waktu berangkat harus berupa "sekarang" atau "jadwal".',
        });
      }
      cleanWaktuBerangkat = normalizedWaktu;
    }

    let cleanJadwalAt = null;
    if (hasValue(jadwal_at)) {
      const parsedDate = Date.parse(jadwal_at);
      if (Number.isNaN(parsedDate)) {
        return res.status(400).json({
          success: false,
          message: 'Format waktu jadwal_at tidak valid.',
        });
      }
      cleanJadwalAt = new Date(parsedDate).toISOString();
    }

    if (cleanWaktuBerangkat === 'jadwal') {
      if (!cleanJadwalAt || Date.parse(cleanJadwalAt) < Date.now() - 60 * 1000) {
        return res.status(400).json({
          success: false,
          message: 'Waktu jadwal harus diisi dan tidak boleh di masa lalu.',
        });
      }
    }

    const cleanNamaBarang = hasValue(nama_barang) ? String(nama_barang).trim() : null;

    let cleanPerkiraanHargaBarang = null;
    if (hasValue(perkiraan_harga_barang)) {
      const parsedHargaBarang = Number(perkiraan_harga_barang);
      if (Number.isNaN(parsedHargaBarang) || parsedHargaBarang < 0) {
        return res.status(400).json({
          success: false,
          message: 'Perkiraan harga barang tidak valid.',
        });
      }
      cleanPerkiraanHargaBarang = parsedHargaBarang;
    }

    // Ongkir wajib valid (tidak lagi diam-diam jadi 5000)
    const parsedOngkir = Number(ongkir);
    if (
      !Number.isInteger(parsedOngkir) ||
      parsedOngkir < MIN_ONGKIR ||
      parsedOngkir > MAX_ONGKIR
    ) {
      return res.status(400).json({
        success: false,
        message: `Ongkir harus berupa angka bulat antara Rp ${MIN_ONGKIR.toLocaleString('id-ID')} dan Rp ${MAX_ONGKIR.toLocaleString('id-ID')}.`,
      });
    }

    // Batasi jumlah order aktif per pemesan (anti spam)
    const { count: activeCount, error: countError } = await supabase
      .from('orders')
      .select('id', { count: 'exact', head: true })
      .eq('customer_id', req.user.id)
      .in('status', ACTIVE_STATUSES);

    if (countError) throw countError;

    if ((activeCount || 0) >= MAX_ACTIVE_ORDERS_PER_CUSTOMER) {
      return res.status(400).json({
        success: false,
        message: `Maksimal ${MAX_ACTIVE_ORDERS_PER_CUSTOMER} pesanan aktif sekaligus. Selesaikan atau batalkan pesanan lain dulu.`,
      });
    }

    const { data, error } = await supabase
      .from('orders')
      .insert([
        {
          customer_id: req.user.id,
          jenis_layanan,
          titik_jemput: titik_jemput.trim(),
          titik_antar: titik_antar.trim(),
          catatan: typeof catatan === 'string' ? catatan.trim() : '',
          ongkir: parsedOngkir,
          status: 'pending',
          jenis_kendaraan: cleanJenisKendaraan,
          jumlah_penumpang: cleanJumlahPenumpang,
          waktu_berangkat: cleanWaktuBerangkat,
          jadwal_at: cleanJadwalAt,
          nama_barang: cleanNamaBarang,
          perkiraan_harga_barang: cleanPerkiraanHargaBarang,
        },
      ])
      .select(ORDER_SELECT_QUERY)
      .single();

    if (error) throw error;

    return res.status(201).json({
      success: true,
      message: 'Pesanan berhasil dibuat! Menunggu driver mengambil order.',
      data,
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal membuat pesanan.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// GET /api/orders (Customer melihat pesanannya; Driver melihat order pending + order miliknya)
export const getOrders = async (req, res) => {
  try {
    // Order pending "Sekarang" yang tidak diambil siapa pun otomatis kedaluwarsa
    try {
      const cutoff = new Date(
        Date.now() - PENDING_EXPIRE_MINUTES * 60 * 1000
      ).toISOString();
      await supabase
        .from('orders')
        .update({ status: 'cancelled', updated_at: new Date().toISOString() })
        .eq('status', 'pending')
        .lt('created_at', cutoff)
        .or('waktu_berangkat.is.null,waktu_berangkat.neq.jadwal');

      // Order "Jadwal" kedaluwarsa bila waktunya sudah lewat > 30 menit dan belum diambil
      const jadwalCutoff = new Date(
        Date.now() - JADWAL_EXPIRE_MINUTES * 60 * 1000
      ).toISOString();
      await supabase
        .from('orders')
        .update({ status: 'cancelled', updated_at: new Date().toISOString() })
        .eq('status', 'pending')
        .eq('waktu_berangkat', 'jadwal')
        .lt('jadwal_at', jadwalCutoff);
    } catch {
      // Kegagalan housekeeping tidak boleh menggagalkan daftar order
    }

    let query = supabase
      .from('orders')
      .select(ORDER_SELECT_QUERY)
      .order('created_at', { ascending: false });

    if (req.user.role === 'customer') {
      query = query.eq('customer_id', req.user.id);
    } else if (req.user.role === 'driver') {
      query = query.or(`status.eq.pending,driver_id.eq.${req.user.id}`);
    }

    const { data, error } = await query;

    if (error) throw error;

    return res.status(200).json({
      success: true,
      data: (data || []).map((order) => hideCustomerContact(order, req.user)),
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal mengambil daftar pesanan.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// PATCH /api/orders/:id/accept (Driver mengambil pesanan pending)
export const acceptOrder = async (req, res) => {
  try {
    if (req.user.role !== 'driver') {
      return res.status(403).json({
        success: false,
        message: 'Hanya akun dengan role Driver yang dapat mengambil pesanan.',
      });
    }

    const { id } = req.params;

    const { data: order, error: findError } = await supabase
      .from('orders')
      .select('id, status, driver_id, jenis_layanan, jenis_kendaraan, jumlah_penumpang')
      .eq('id', id)
      .maybeSingle();

    if (findError) throw findError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (order.status !== 'pending' || order.driver_id) {
      return res.status(409).json({
        success: false,
        message: 'Pesanan ini sudah diambil oleh driver lain atau tidak lagi tersedia.',
      });
    }

    // Validasi: online, kendaraan lengkap & cocok, kapasitas cukup, tidak ada trip aktif
    const eligibility = await checkDriverEligibility(req.user.id, order);
    if (!eligibility.ok) {
      return res.status(eligibility.status).json({
        success: false,
        message: eligibility.message,
      });
    }

    // Update bersyarat untuk mencegah race condition antar driver
    const { data, error } = await supabase
      .from('orders')
      .update({
        driver_id: req.user.id,
        status: 'accepted',
        updated_at: new Date().toISOString(),
      })
      .eq('id', id)
      .eq('status', 'pending')
      .is('driver_id', null)
      .select(ORDER_SELECT_QUERY)
      .maybeSingle();

    if (error) throw error;

    if (!data) {
      return res.status(409).json({
        success: false,
        message: 'Pesanan ini sudah diambil oleh driver lain atau tidak lagi tersedia.',
      });
    }

    return res.status(200).json({
      success: true,
      message: 'Pesanan berhasil diambil! Segera hubungi customer via WhatsApp.',
      data,
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal mengambil pesanan.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// PATCH /api/orders/:id/status (Update status perjalanan / pembatalan)
export const updateOrderStatus = async (req, res) => {
  try {
    const { id } = req.params;
    const { status } = req.body;

    const allowedStatuses = ['on_progress', 'completed', 'cancelled'];
    if (!allowedStatuses.includes(status)) {
      return res.status(400).json({
        success: false,
        message: 'Status tidak valid. Pilihan: on_progress, completed, cancelled.',
      });
    }

    const { data: order, error: findError } = await supabase
      .from('orders')
      .select('*')
      .eq('id', id)
      .maybeSingle();

    if (findError) throw findError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    const ALLOWED_TRANSITIONS = {
      accepted: ['on_progress', 'cancelled'],
      on_progress: ['completed', 'cancelled'],
      pending: ['cancelled'],
    };

    if (!(ALLOWED_TRANSITIONS[order.status] || []).includes(status)) {
      return res.status(400).json({
        success: false,
        message: `Pesanan berstatus "${order.status}" tidak dapat diubah menjadi "${status}".`,
      });
    }

    if (status === 'cancelled') {
      // Pemesan boleh membatalkan selama belum mulai diantar (pending / accepted)
      const isOwnerCustomer =
        order.customer_id === req.user.id &&
        ['pending', 'accepted'].includes(order.status);
      const isAssignedDriver = order.driver_id === req.user.id;

      if (!isOwnerCustomer && !isAssignedDriver) {
        return res.status(403).json({
          success: false,
          message: 'Pesanan tidak dapat dibatalkan pada tahap ini.',
        });
      }
    } else if (order.driver_id !== req.user.id) {
      return res.status(403).json({
        success: false,
        message: 'Anda bukan driver yang bertugas untuk pesanan ini.',
      });
    }

    const { data, error } = await supabase
      .from('orders')
      .update({
        status,
        updated_at: new Date().toISOString(),
      })
      .eq('id', id)
      .eq('status', order.status)
      .select(ORDER_SELECT_QUERY)
      .maybeSingle();

    if (error) throw error;

    if (!data) {
      return res.status(409).json({
        success: false,
        message: 'Status pesanan baru saja berubah. Muat ulang lalu coba lagi.',
      });
    }

    return res.status(200).json({
      success: true,
      message: `Status pesanan diperbarui menjadi "${status}".`,
      data,
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal memperbarui status pesanan.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// DELETE /api/orders/:id (Customer menghapus riwayat pesanan yang pending/cancelled)
export const deleteOrder = async (req, res) => {
  try {
    const { id } = req.params;

    const { data: order, error: findError } = await supabase
      .from('orders')
      .select('id, customer_id, status')
      .eq('id', id)
      .maybeSingle();

    if (findError) throw findError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (order.customer_id !== req.user.id) {
      return res.status(403).json({
        success: false,
        message: 'Anda tidak berhak menghapus pesanan ini.',
      });
    }

    if (!['pending', 'cancelled'].includes(order.status)) {
      return res.status(400).json({
        success: false,
        message: 'Pesanan yang sedang berjalan atau sudah selesai tidak dapat dihapus.',
      });
    }

    const { error } = await supabase.from('orders').delete().eq('id', id);

    if (error) throw error;

    return res.status(200).json({
      success: true,
      message: 'Pesanan berhasil dihapus.',
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal menghapus pesanan.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

import supabase from '../config/supabase.js';
import { checkDriverEligibility } from '../utils/driverEligibility.js';

const OFFER_SELECT_QUERY = `
  *,
  driver:users!driver_id(id, nama, nim, no_wa, jenis_kendaraan, plat_nomor, kapasitas_penumpang, is_online)
`;

const ORDER_SELECT_QUERY = `
  *,
  customer:users!customer_id(id, nama, nim, no_wa),
  driver:users!driver_id(id, nama, nim, no_wa, jenis_kendaraan, plat_nomor, kapasitas_penumpang, is_online)
`;

const MIN_PRICE = 2000;
const MAX_PRICE = 1000000;
const MAX_PENUMPANG = 8;
const MAX_NOTE_LENGTH = 200;

const ORDER_FOR_OFFER_FIELDS =
  'id, customer_id, status, jenis_layanan, jenis_kendaraan, jumlah_penumpang';

// Kontak driver baru terlihat setelah tawaran disepakati
// (sama seperti kontak pemesan yang disembunyikan dari driver sebelum order diterima)
const sanitizeOffer = (offer) => {
  if (!offer || !offer.driver || offer.status === 'accepted') return offer;
  return {
    ...offer,
    driver: { ...offer.driver, nim: null, no_wa: null },
  };
};

// GET /api/orders/:id/offers
export const getOrderOffers = async (req, res) => {
  try {
    const { id } = req.params;

    const { data: order, error: orderError } = await supabase
      .from('orders')
      .select('id, customer_id, driver_id, status')
      .eq('id', id)
      .maybeSingle();

    if (orderError) throw orderError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (
      req.user.role === 'customer' &&
      order.customer_id !== req.user.id
    ) {
      return res.status(403).json({
        success: false,
        message: 'Anda tidak berhak melihat tawaran untuk pesanan ini.',
      });
    }

    let query = supabase
      .from('offers')
      .select(OFFER_SELECT_QUERY)
      .eq('order_id', id)
      .order('created_at', { ascending: false });

    if (req.user.role === 'driver') {
      query = query.eq('driver_id', req.user.id);
    }

    const { data, error } = await query;

    if (error) throw error;

    return res.status(200).json({
      success: true,
      data: (data || []).map(sanitizeOffer),
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal mengambil daftar tawaran.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// POST /api/orders/:id/offers
export const createOrderOffer = async (req, res) => {
  try {
    const { id } = req.params;
    const { offered_price, driver_id, adjusted_passengers, note } = req.body;

    const parsedPrice = Number(offered_price);
    if (
      !Number.isInteger(parsedPrice) ||
      parsedPrice < MIN_PRICE ||
      parsedPrice > MAX_PRICE
    ) {
      return res.status(400).json({
        success: false,
        message: `Nominal tawaran harus berupa angka bulat antara Rp ${MIN_PRICE.toLocaleString('id-ID')} dan Rp ${MAX_PRICE.toLocaleString('id-ID')}.`,
      });
    }

    const cleanNote = note ? String(note).trim() : '';
    if (cleanNote.length > MAX_NOTE_LENGTH) {
      return res.status(400).json({
        success: false,
        message: `Catatan maksimal ${MAX_NOTE_LENGTH} karakter.`,
      });
    }

    const { data: order, error: orderError } = await supabase
      .from('orders')
      .select(ORDER_FOR_OFFER_FIELDS)
      .eq('id', id)
      .maybeSingle();

    if (orderError) throw orderError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (order.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'Negosiasi hanya dapat dilakukan pada pesanan yang masih menunggu driver.',
      });
    }

    const isDriver = req.user.role === 'driver';
    const offeredBy = isDriver ? 'driver' : 'customer';
    let targetDriverId = null;
    let cleanPassengers = null;

    // Penyesuaian jumlah penumpang hanya boleh dari pemesan, dan hanya untuk AnJem
    if (
      !isDriver &&
      order.jenis_layanan === 'anjem' &&
      adjusted_passengers !== undefined &&
      adjusted_passengers !== null &&
      adjusted_passengers !== ''
    ) {
      const parsedPax = Number(adjusted_passengers);
      if (
        !Number.isInteger(parsedPax) ||
        parsedPax < 1 ||
        parsedPax > MAX_PENUMPANG
      ) {
        return res.status(400).json({
          success: false,
          message: `Jumlah penumpang harus antara 1 hingga ${MAX_PENUMPANG} orang.`,
        });
      }
      cleanPassengers = parsedPax;
    }

    if (isDriver) {
      targetDriverId = req.user.id;

      // Driver wajib ON, punya data kendaraan, dan kendaraannya cocok.
      // Kapasitas & trip aktif baru dicek saat tawaran disepakati.
      const eligibility = await checkDriverEligibility(targetDriverId, order, {
        skipCapacity: true,
        skipActiveTrip: true,
      });
      if (!eligibility.ok) {
        return res.status(eligibility.status).json({
          success: false,
          message: eligibility.message,
        });
      }
    } else {
      if (order.customer_id !== req.user.id) {
        return res.status(403).json({
          success: false,
          message: 'Anda bukan pembuat pesanan ini.',
        });
      }

      targetDriverId = driver_id;
      if (!targetDriverId || typeof targetDriverId !== 'string') {
        return res.status(400).json({
          success: false,
          message: 'driver_id wajib disertakan saat pemesan menawar balik.',
        });
      }

      // Pemesan hanya boleh menawar balik ke driver yang sudah menawar lebih dulu
      const { data: driverOffer, error: driverOfferError } = await supabase
        .from('offers')
        .select('id')
        .eq('order_id', id)
        .eq('driver_id', targetDriverId)
        .eq('offered_by', 'driver')
        .limit(1);

      if (driverOfferError) throw driverOfferError;

      if (!driverOffer || driverOffer.length === 0) {
        return res.status(400).json({
          success: false,
          message: 'Driver ini belum mengirim tawaran untuk pesananmu.',
        });
      }

      const eligibility = await checkDriverEligibility(targetDriverId, order, {
        passengers: cleanPassengers ?? order.jumlah_penumpang,
        skipActiveTrip: true,
      });
      if (!eligibility.ok) {
        return res.status(eligibility.status).json({
          success: false,
          message: eligibility.message,
        });
      }
    }

    const { data: createdOffer, error: insertError } = await supabase
      .from('offers')
      .insert([
        {
          order_id: id,
          driver_id: targetDriverId,
          offered_by: offeredBy,
          offered_price: parsedPrice,
          adjusted_passengers: cleanPassengers,
          note: cleanNote || null,
          status: 'pending',
        },
      ])
      .select(OFFER_SELECT_QUERY)
      .single();

    if (insertError) throw insertError;

    // Tandai tawaran pending sebelumnya (selain yang baru) sebagai superseded
    await supabase
      .from('offers')
      .update({
        status: 'superseded',
        updated_at: new Date().toISOString(),
      })
      .eq('order_id', id)
      .eq('driver_id', targetDriverId)
      .eq('status', 'pending')
      .neq('id', createdOffer.id);

    // Catatan: tabel orders TIDAK diubah saat tawar-menawar.
    // Harga & jumlah penumpang final baru diterapkan ketika tawaran disepakati (accept).

    return res.status(201).json({
      success: true,
      message: 'Penawaran harga berhasil dikirim.',
      data: sanitizeOffer(createdOffer),
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal mengirim penawaran harga.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// PATCH /api/orders/:id/offers/:offerId/accept
export const acceptOrderOffer = async (req, res) => {
  try {
    const { id, offerId } = req.params;

    const { data: order, error: orderError } = await supabase
      .from('orders')
      .select(ORDER_FOR_OFFER_FIELDS)
      .eq('id', id)
      .maybeSingle();

    if (orderError) throw orderError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (order.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'Pesanan ini sudah diproses atau tidak lagi tersedia.',
      });
    }

    const { data: offer, error: offerError } = await supabase
      .from('offers')
      .select('*')
      .eq('id', offerId)
      .eq('order_id', id)
      .maybeSingle();

    if (offerError) throw offerError;

    if (!offer) {
      return res.status(404).json({
        success: false,
        message: 'Tawaran tidak ditemukan.',
      });
    }

    if (offer.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'Tawaran ini sudah tidak berlaku (sudah diproses atau digantikan).',
      });
    }

    // Validasi: pihak yang menyetujui harus lawan dari pembuat tawaran
    if (offer.offered_by === 'driver' && order.customer_id !== req.user.id) {
      return res.status(403).json({
        success: false,
        message: 'Hanya pemesan yang dapat menyetujui tawaran driver.',
      });
    }

    if (offer.offered_by === 'customer' && offer.driver_id !== req.user.id) {
      return res.status(403).json({
        success: false,
        message: 'Hanya driver terkait yang dapat menyetujui tawaran pemesan.',
      });
    }

    const finalPassengers =
      order.jenis_layanan === 'anjem'
        ? offer.adjusted_passengers || order.jumlah_penumpang
        : order.jumlah_penumpang;

    // Driver harus masih memenuhi syarat: ON, kendaraan cocok & muat, tidak ada trip aktif
    const eligibility = await checkDriverEligibility(offer.driver_id, order, {
      passengers: finalPassengers,
    });
    if (!eligibility.ok) {
      const isDriverSelf = req.user.id === offer.driver_id;
      return res.status(eligibility.status).json({
        success: false,
        message: isDriverSelf
          ? eligibility.message
          : 'Driver ini sedang tidak dapat menerima order (offline, kendaraan tidak sesuai, atau sedang menjalankan trip lain). Coba tawaran lain.',
      });
    }

    // Proses penerimaan dijalankan dalam satu transaksi PostgreSQL melalui RPC.
    // RPC mengunci baris order dan memperbarui order + seluruh status offer secara atomik.
    const { data: acceptance, error: acceptanceError } = await supabase.rpc(
      'accept_order_offer_atomic',
      {
        p_order_id: id,
        p_offer_id: offerId,
        p_actor_id: req.user.id,
      }
    );

    if (acceptanceError) throw acceptanceError;

    if (!acceptance?.success) {
      return res.status(acceptance?.status || 409).json({
        success: false,
        message: acceptance?.message || 'Pesanan atau tawaran sudah berubah. Muat ulang lalu coba lagi.',
      });
    }

    // Ambil ulang relasi customer/driver agar bentuk respons tetap sama seperti API sebelumnya.
    const { data: updatedOrder, error: fetchUpdatedOrderError } = await supabase
      .from('orders')
      .select(ORDER_SELECT_QUERY)
      .eq('id', id)
      .single();

    if (fetchUpdatedOrderError) throw fetchUpdatedOrderError;

    return res.status(200).json({
      success: true,
      message: 'Tawaran disepakati! Pesanan kini aktif.',
      data: updatedOrder,
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal menyetujui tawaran.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

// PATCH /api/orders/:id/offers/:offerId/reject
export const rejectOrderOffer = async (req, res) => {
  try {
    const { id, offerId } = req.params;

    const { data: order, error: orderError } = await supabase
      .from('orders')
      .select('id, customer_id, status')
      .eq('id', id)
      .maybeSingle();

    if (orderError) throw orderError;

    if (!order) {
      return res.status(404).json({
        success: false,
        message: 'Pesanan tidak ditemukan.',
      });
    }

    if (order.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'Pesanan ini sudah diproses atau tidak lagi tersedia.',
      });
    }

    const { data: offer, error: offerError } = await supabase
      .from('offers')
      .select('id, order_id, driver_id, offered_by, status')
      .eq('id', offerId)
      .eq('order_id', id)
      .maybeSingle();

    if (offerError) throw offerError;

    if (!offer) {
      return res.status(404).json({
        success: false,
        message: 'Tawaran tidak ditemukan.',
      });
    }

    if (offer.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'Tawaran ini sudah tidak berlaku (sudah diproses atau digantikan).',
      });
    }

    // Yang boleh menolak adalah lawan dari pembuat tawaran
    const isAllowed =
      (offer.offered_by === 'driver' && order.customer_id === req.user.id) ||
      (offer.offered_by === 'customer' && offer.driver_id === req.user.id);

    if (!isAllowed) {
      return res.status(403).json({
        success: false,
        message: 'Anda tidak berhak menolak tawaran ini.',
      });
    }

    const { data: rejected, error: rejectError } = await supabase
      .from('offers')
      .update({ status: 'rejected', updated_at: new Date().toISOString() })
      .eq('id', offerId)
      .eq('status', 'pending')
      .select(OFFER_SELECT_QUERY)
      .maybeSingle();

    if (rejectError) throw rejectError;

    if (!rejected) {
      return res.status(409).json({
        success: false,
        message: 'Tawaran ini baru saja berubah. Muat ulang lalu coba lagi.',
      });
    }

    return res.status(200).json({
      success: true,
      message: 'Tawaran ditolak.',
      data: sanitizeOffer(rejected),
    });
  } catch (error) {
    return res.status(500).json({
      success: false,
      message: 'Gagal menolak tawaran.',
      error: process.env.NODE_ENV === 'production' ? undefined : error.message,
    });
  }
};

export default function Button({
  children,
  to,
  variant = 'primary',
  height = 48,
  disabled = false,
  type = 'button',
  style,
  className = '',
  ...props
}) {
  const activeVariant = disabled ? 'disabled' : variant;
  const variantStyle = BUTTON_VARIANTS[activeVariant] || BUTTON_VARIANTS.primary;

  const baseStyle = {
    height: `${height}px`,
    borderRadius: '14px',

    width: '100%',
    textDecoration: 'none',
    boxSizing: 'border-box',
    ...variantStyle,
    ...style,
  };

  if (to && !disabled) {
    return (
      <Link to={to} className={className} style={baseStyle} {...props}>
        {children}
      </Link>
    );

    <button
      type={type}
      disabled={disabled}
      className={className}
      style={baseStyle}
      {...props}
    >

    </button>
  );
}

import { Navigate } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

export default function ProtectedRoute({ children, allowedRole }) {
  const { user, loading } = useAuth();

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-100">
        <div className="text-sm font-medium text-slate-500 animate-pulse">
          Memuat sesi AnjemDips...
        </div>
      </div>
    );
  }

  if (!user) {
    return <Navigate to="/login" replace />;
  }

  if (allowedRole && user.role !== allowedRole) {
    return <Navigate to={user.role === 'driver' ? '/driver' : '/customer'} replace />;
  }

  return children;
}
